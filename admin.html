Supabase Table & Admin Policies SQL

Run this in your Supabase Dashboard → SQL Editor so your Admin Panel can view and delete messages: 

create table if not exists public.contact_messages (
  id bigint generated by default as identity primary key,
  name text not null,
  email text not null,
  phone text,
  service text,
  message text not null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table public.contact_messages enable row level security;

drop policy if exists "Allow anonymous inserts" on public.contact_messages;
drop policy if exists "Allow public read" on public.contact_messages;
drop policy if exists "Allow public delete" on public.contact_messages;

create policy "Allow anonymous inserts" on public.contact_messages for insert to anon with check (true);
create policy "Allow public read" on public.contact_messages for select to anon using (true);
create policy "Allow public delete" on public.contact_messages for delete to anon using (true);
